This is the dynamic version that loads events from Supabase and uses your $2/$5 Stripe links.

jsx
import { useEffect, useState } from 'react';
import { supabase } from '../../lib/supabaseClient';
import { buildStripeUrl } from '../../lib/buildStripeUrl';

export default function EventsPage() {
  const [events, setEvents] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function loadEvents() {
      const { data, error } = await supabase
        .from('events')
        .select('*')
        .order('created_at', { ascending: false });

      if (error) {
        console.error('Error loading events:', error);
      } else {
        setEvents(data || []);
      }

      setLoading(false);
    }

    loadEvents();
  }, []);

  if (loading) {
    return (
      <div className="p-6 text-center text-gray-300">
        Loading events…
      </div>
    );
  }

  return (
    <div className="p-6 max-w-6xl mx-auto">
      <h1 className="text-3xl font-bold text-white mb-6">Live Events</h1>

      {events.length === 0 && (
        <div className="text-gray-400 text-center mt-10">
          No events yet. Be the first to create one!
        </div>
      )}

      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
        {events.map(event => (
          <div
            key={event.id}
            className="bg-[#111] border border-gray-800 rounded-xl p-5 shadow-lg"
          >
            <h2 className="text-xl font-semibold text-white mb-2">
              {event.title}
            </h2>

            <p className="text-gray-400 text-sm mb-1">
              📅 {event.date}
            </p>
            <p className="text-gray-400 text-sm mb-3">
              📍 {event.location}
            </p>

            <p className="text-gray-300 text-sm mb-4">
              {event.description}
            </p>

            <div className="flex items-center justify-between mt-4">
              <span className="text-white font-bold text-lg">
                ${event.price}
              </span>

              <a
                href={buildStripeUrl(event.price, 'event')}
                target="_blank"
                rel="noopener noreferrer"
                className="bg-blue-600 hover:bg-blue-700 text-white px-4 py-2 rounded-lg text-sm font-semibold"
              >
                Buy Ticket
              </a>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}
