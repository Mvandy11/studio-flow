const subscription = event.data.object;
const status = subscription.status;

console.log("Stripe subscription status:", status);

let membershipStatus = "inactive";

// Treat trialing as active
if (["active", "trialing", "past_due"].includes(status)) {
    membershipStatus = "active";
}

// Treat canceled, unpaid, incomplete, incomplete_expired as inactive
if (["canceled", "unpaid", "incomplete", "incomplete_expired"].includes(status)) {
    membershipStatus = "inactive";
}

// Update Supabase
const { error } = await supabaseAdmin
  .from("profiles")
  .update({ membership_status: membershipStatus })
  .eq("stripe_customer_id", subscription.customer);

if (error) {
  console.error("Supabase membership update error:", error);
} else {
  console.log("Membership updated to:", membershipStatus);
}
